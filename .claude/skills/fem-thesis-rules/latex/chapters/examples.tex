\section{Figure}
A figure is introduced in the text before it appears (Figure~\ref{fig:example}).

\begin{figure}[ht]
  \centering
  \framebox[0.6\textwidth]{\rule{0pt}{3cm}Figure content}
  \caption{Isosceles triangle}\label{fig:example}
  \figsource{\cite[123]{Nowak2007}; \cite[22]{Kowalski2009}}
\end{figure}

\section{Table}
The table title is placed above the table and the source below it
(Table~\ref{tab:example}).

\begin{table}[ht]
  \caption{Analysis of employment size}\label{tab:example}
  \centering
  \begin{tabular}{llll}
    \toprule
    Column 1 & Column 2 & Column 3 & Column 4 \\
    \midrule
    a & b & c & d \\
    \bottomrule
  \end{tabular}
  \tabsource{own elaboration}
\end{table}

\section{Formula}
Formulas are centred and numbered on the right, with a source:
\begin{equation}
  (a+b)^2 = a^2 + 2ab + b^2
  \label{eq:example}
\end{equation}
\eqsource{own elaboration}
